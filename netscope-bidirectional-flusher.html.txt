    class TracePluginManager {
      constructor() {
        this.plugins = new Map();
      }
      register(plugin) {
        this.plugins.set(plugin.id, { ...plugin, enabled: true });
      }
      toggle(id) {
        const p = this.plugins.get(id);
        if (p) p.enabled = !p.enabled;
      }
      async emit(hook, data) {
        for (const plugin of this.plugins.values()) {
          if (plugin.enabled && typeof plugin[hook] === "function") {
            try {
              await plugin[hook](data);
            } catch (err) {
              logEvent(`[Plugin:${plugin.id}] ${err.message}`);
            }
          }
        }
      }
    }


    const plugins = new TracePluginManager();

    /**
     * 3. FULL OUT -> IN -> OUT -> IN BIDIRECTIONAL TRACING
     */
    async function traceBidirectionalExchange(url, bypassCache = false) {
      const app = getActiveApp();
      const port = extractPort(url);
      const portEntry = monitoredPorts.find((p) => p.port === port);


      // Block if port was flushed
      if (portEntry && portEntry.flushed) {
        const blockedPkt = createPacketRecord({
          appId: app.id,
          direction: "OUT",
          hop: "OUT-1 (BLOCKED)",
          url,
          port,
          state: "PORT FLUSHED",
          status: 403,
          sizeBytes: 0,
          durationMs: 0,
          unwanted: true
        });
        packetLog.unshift(blockedPkt);
        await plugins.emit("onPacket", blockedPkt);
        renderAll();
        return;
      }


      // HOP 1: OUTBOUND REQUEST FRAME (OUT ->)
      const outStart = performance.now();
      const outPacket = createPacketRecord({
        appId: app.id,
        direction: "OUT",
        hop: bypassCache ? "OUT-1 (LIVE WIRE)" : "OUT-1 (CACHE CHECK)",
        url,
        port,
        state: bypassCache ? "SYN / GET" : "LOOKUP",
        status: 0,
        sizeBytes: new Blob([url]).size + 128,
        durationMs: 1,
        unwanted: false
      });
      packetLog.unshift(outPacket);
      await plugins.emit("onPacket", outPacket);


      // CACHE OR NETWORK RESOLUTION
      const bucket = `netscope-v2:${app.id}`;
      const hasCacheApi = "caches" in window;
      let cachedHit = false;
      let bodyText = "";
      let status = 200;


      if (app.cacheEnabled && !bypassCache) {
        if (hasCacheApi) {
          const cache = await caches.open(bucket);
          const match = await cache.match(url);
          if (match) {
            cachedHit = true;
            status = match.status;
            bodyText = await match.text();
          }
        } else if (memoryCache.has(`${app.id}:${url}`)) {
          cachedHit = true;
          const item = memoryCache.get(`${app.id}:${url}`);
          status = item.status;
          bodyText = item.body;
        }
      }


      if (!cachedHit) {
        try {
          const res = await fetch(url, { cache: "no-store" });
          status = res.status;
          bodyText = await res.text();
          if (app.cacheEnabled && res.ok) {
            if (hasCacheApi) {
              const cache = await caches.open(bucket);
              await cache.put(url, new Response(bodyText, { status }));
            } else {
              memoryCache.set(`${app.id}:${url}`, { status, body: bodyText });
            }
          }
        } catch (err) {
          status = 502;
          bodyText = err.message;
        }
      }


      const rtt = Math.max(1, Math.round(performance.now() - outStart));


      // HOP 2: INBOUND RESPONSE FRAME (<- IN)
      const inPacket = createPacketRecord({
        appId: app.id,
        direction: "IN",
        hop: cachedHit ? "IN-1 (CACHE HIT)" : "IN-1 (NET PAYLOAD)",
        url,
        port,
        state: cachedHit ? "CACHE HIT" : `HTTP ${status}`,
        status,
        sizeBytes: new Blob([bodyText]).size,
        durationMs: cachedHit ? Math.min(rtt, 3) : rtt,
        unwanted: status >= 400
      });


      const verdict = evaluatePacketPolicy(inPacket);
      if (verdict && verdict.unwanted) {
        inPacket.unwanted = true;
        inPacket.state = verdict.reason || "FLAGGED";
      }


      packetLog.unshift(inPacket);
      await plugins.emit("onPacket", inPacket);


      // Keep buffer bounded
      if (packetLog.length > 80) packetLog.length = 80;
      renderAll();
    }


    function createPacketRecord({ appId, direction, hop, url, port, state, status, sizeBytes, durationMs, unwanted }) {
      return {
        id: `pkt-${Date.now()}-${Math.random().toString(36).slice(2, 7)}`,
        seq: packetSeq++,
        appId,
        direction,
        hop,
        url,
        shortUrl: url.replace(/^https?:\/\//, "").slice(0, 38) + (url.length > 46 ? "..." : ""),
        port,
        state,
        status,
        sizeBytes,
        durationMs,
        unwanted,
        timestamp: new Date().toLocaleTimeString()
      };
    }


    /**
     * 4. SIMULATE UNWANTED PACKETS & FLUSH ENGINE
     */
    async function simulateUnwantedPacket() {
      const app = getActiveApp();
      const samples = [
        { dir: "IN", url: "http://127.0.0.1:8080/telemetry/beacon-noise", port: 8080, status: 429, state: "UNWANTED BEACON" },
        { dir: "OUT", url: "http://127.0.0.1:3000/debug/unused-poll", port: 3000, status: 404, state: "STALE POLL" },
        { dir: "IN", url: "https://tracker.invalid.local/pixel.gif", port: 443, status: 403, state: "UNSOLICITED IN" }
      ];
      const pick = samples[Math.floor(Math.random() * samples.length)];
      const pkt = createPacketRecord({
        appId: app.id,
        direction: pick.dir,
        hop: `${pick.dir}-NOISE`,
        url: pick.url,
        port: pick.port,
        state: pick.state,
        status: pick.status,
        sizeBytes: 512,
        durationMs: 12,
        unwanted: true
      });
      packetLog.unshift(pkt);
      await plugins.emit("onPacket", pkt);
      renderAll();
    }


    async function flushPackets(predicate, reason) {
      const before = packetLog.length;
      const kept = [];
      const removed = [];


      for (const pkt of packetLog) {
        if (predicate(pkt)) {
          removed.push(pkt);
        } else {
          kept.push(pkt);
        }
      }


      packetLog = kept;
      const count = before - kept.length;
      totalFlushedPackets += count;
      await plugins.emit("onFlush", { count, reason, removed });
      renderAll();
    }


    function flushSinglePacket(id) {
      flushPackets((p) => p.id === id, "Manual single packet flush");
    }


    /**
     * 5. PORT PROBE & PORT FLUSHER
     */
    async function probeSinglePort(portObj) {
      if (portObj.flushed) {
        portObj.status = "FLUSHED";
        portObj.isOpen = false;
        return portObj;
      }


      const controller = new AbortController();
      const timer = setTimeout(() => controller.abort(), 600);
      const start = performance.now();


      try {
        await fetch(`http://127.0.0.1:${portObj.port}/`, { mode: "no-cors", signal: controller.signal });
        portObj.isOpen = true;
        portObj.status = `OPEN (${Math.round(performance.now() - start)}ms)`;
      } catch (err) {
        portObj.isOpen = false;
        portObj.status = err.name === "AbortError" ? "TIMEOUT" : "CLOSED";
      } finally {
        clearTimeout(timer);
      }
      return portObj;
    }


    async function sweepAllPorts() {
      for (const p of monitoredPorts) {
        await probeSinglePort(p);
      }
      logEvent("Completed local port sweep across monitored ports.");
      renderAll();
    }


    async function toggleFlushPort(portNum) {
      const target = monitoredPorts.find((p) => p.port === portNum);
      if (!target) return;
      target.flushed = !target.flushed;
      if (target.flushed) {
        target.isOpen = false;
        target.status = "FLUSHED";
        await plugins.emit("onPortFlush", target);
        // Also flush any packets currently associated with that port
        await flushPackets((pkt) => pkt.port === portNum, `Port :${portNum} flushed`);
      } else {
        target.status = "UNPROBED";
        logEvent(`Restored port :${portNum} (${target.service})`);
        renderAll();
      }
    }

      const inCount = appPackets.filter((p) => p.direction === "IN").length;

        if (dirFilter === "IN" && p.direction !== "IN") return false;

          )
          .join("");
      }

      portGrid.innerHTML = monitoredPorts
        .map(

              </span>
            </div>

        .join("");
